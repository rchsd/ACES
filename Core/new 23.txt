__NDIM __SPLIT_ARGS(2) split_faces( const Polygon<2, __ndim> & pgon, const __VECTOR & normal, const Real & coef )
{
	/* ----------------------------------------- Declare Local Variables ----------------------------------------- */

	Polygon<2, __ndim>	result;							/* Output polygon */
	Collection<bool>	over_plane;						/* Mask indicating which output faces are over plane */
	Collection<Index>	original_faces;					/* Correlates input and output face */
	Collection<Real>	vcoefs(pgon.nverts());			/* Stores signed distnaces between vertices and plane */
	Index				nfaces = pgon.nfaces();			/* Number of output faces */
	Index				nverts = pgon.nverts();			/* Number of polygon vertices */
	Vector<2>			signs;							/* Signs of vertices */

	/* ----------------------------------- Calculate Vertex To Plane Distances ----------------------------------- */
	
	/* Calculate planar coefficients of vertices */
	for_each_item(vertex __in__ pgon.vertices(), vcoef __in__ vcoefs)
	{
		/* "Snap" coefficients to zero if they're close to avoid making really small faces */
		if (fabs((*vcoef = *vertex * normal - coef) / coef) < __INTERSECT_EPS) 
			*vcoef = 0;

	} end_for;

	/* -------------------------------------- Locate Edges That Cross Plane -------------------------------------- */

	/* Calculate number of new faces and vertices needed */
	for_each_item(face __in__ pgon.faces())
	{
		/* Calculate signs */
		signs[0] = sign(vcoefs[(*face)[0]]);
		signs[1] = sign(vcoefs[(*face)[1]]);

		/* Update number of faces */
		nfaces += (signs[0] != signs[1] && signs[0] && signs[1]);

	} end_for;

	/* ------------------------------------------ Create Output Polygon ------------------------------------------ */
	
	/* Create output polygon */
	result = Polygon<2, __ndim>(nfaces, nverts + nfaces - pgon.nfaces());
	
	/* Initialize output vertex iterator */
	auto out_vertex = result.vbegin();
		
	/* Copy vertices from input to output */
	for_each_item(vertex __in__ pgon.vertices())
	{
		/* Copy vertex */
		*(out_vertex++) = (*vertex);

	} end_for;
	
	/* Create face property lists */
	over_plane = Collection<bool>(nfaces);
	original_faces = Collection<Index>(nfaces);
	
	/* Initialize face iterators */
	auto out_face = result.fbegin();
	auto over = over_plane.begin();
	auto original = original_faces.begin();

	/* Copy faces from input to output, splitting faces as necessary */
	for_each_item(face __in__ pgon.faces())
	{
		/* Determine face index */
		Index findex = (Index)(face - pgon.faces().begin());

		/* Calculate signs */
		signs[0] = sign(vcoefs[(*face)[0]]);
		signs[1] = sign(vcoefs[(*face)[1]]);
		
		/* Split face in two */
		if (signs[0] != signs[1] && signs[0] && signs[1])
		{
			/* Save first face */
			(*out_face)[0] = (*face)[0];
			(*(out_face++))[1] = nverts;

			/* Save second face */
			(*out_face)[0] = nverts++;
			(*(out_face++))[1] = (*face)[1];

			/* Save over / under */
			*(over++) = vcoefs[(*face)[0]] > 0;
			*(over++) = vcoefs[(*face)[1]] > 0;

			/* Save original face index */
			*(original++) = findex;
			*(original++) = findex;

			/* Calculate weighted average of edge vertices */
			*(out_vertex++) = weighted_average(pgon((*face)[0]), pgon((*face)[1]), 
				vcoefs[(*face)[1]] / (vcoefs[(*face)[1]] - vcoefs[(*face)[0]]));
		}

		/* If face doesn't need to be split, copy it to output */
		else
		{
			/* Copy face */
			*(out_face++) = (*face);

			/* Save over / under */
			*(over++) = (signs[0] > 0 || signs[1] > 0);

			/* Save original face index */
			*(original++) = findex;
		}

	} end_for;
	
	/* Return output arguments */
	return std :: tuple<Polygon<2, __ndim>, Collection<bool>, Collection<Index>>(std :: move(result), 
		std :: move(over_plane), std :: move(original_faces));

	/* ---------------------------------------------- End of Method ---------------------------------------------- */
}

/* =================================================================================================================== */
__NDIM __SPLIT_ARGS(3) split_faces( const Polygon<3, __ndim> & pgon, const __VECTOR & normal, const Real & coef )
{
	/* ----------------------------------------- Declare Local Variables ----------------------------------------- */

	Polygon<3, __ndim>	out_pgon;						/* Output polygon */
	Collection<bool>	over_plane;						/* Mask indicating which output faces are over plane */
	Collection<Index>	original_faces;					/* Correlates input and output face */
	
	Collection<Real>	vcoefs(pgon.nverts());			/* Stores signed distnaces between vertices and plane */
	Collection<Index>	nsplits(pgon.nfaces(), true);	/* Number of child faces for each input face */
	Faces<3>			splits(pgon.nfaces(), true);	/* Stores information used to make splits */
	Faces<2>			edges(pgon.nfaces() * 3);		/* Stores edges that cross the plane */
	HashTable<Face<2>>	table(edges.nitems());			/* Edge lookup table */
	
	Index				nfaces = pgon.nfaces();			/* Number of output faces */

	/* ----------------------------------- Calculate Vertex To Plane Distances ----------------------------------- */
	
	/* Calculate planar coefficients of vertices */
	for_each_item(vertex __in__ pgon.vertices(), vcoef __in__ vcoefs)
	{
		/* "Snap" coefficients to zero if they're close to avoid making really small faces */
		if (fabs((*vcoef = *vertex * normal - coef) / coef) < __INTERSECT_EPS) 
			*vcoef = 0;

	} end_for;

	/* -------------------------------------- Locate Edges That Cross Plane -------------------------------------- */

	/* Initialize edge iterator */
	auto edge = edges.begin();
	
	/* Locate edges that cross plane */
	for_each_item(face __in__ pgon.faces(), nsplit __in__ nsplits, split __in__ splits)
	{
		/* Define vector to store coefficient signs */
		Real signs[] = {sign(vcoefs[(*face)[0]]), sign(vcoefs[(*face)[1]]), sign(vcoefs[(*face)[2]])};
		
		/* Define variable to store inserted edge */
		Face<2> * inserted;

		/* Check to see if edges cross plane */
		for (Index k, i = 0, j = 1; i < 3; i++, (++j) %= 3)
		{
			/* If edge crosses plane, process it */
			if (signs[i] != signs[j] && signs[i] && signs[j])
			{
				/* Create edge */
				(*edge)[k = (*face)[i] > (*face)[j]] = (*face)[i];
				(*edge)[!k] = (*face)[j];
				
				/* Insert edge into hash table */
				edge += ((inserted = table.insert(edge)) == edge);
				
				/* Add index of new vertex to (*face) helper array */
				(*split)[i] = (Index)(inserted - edges.begin()) + pgon.nverts();

				/* Increment number of splits for current (*face) */
				(*nsplit)++;
			}
		}

		/* Update number of faces */
		nfaces += (*nsplit);

	} end_for;

	/* Eliminate unused memory from edges buffer */
	edges.shrink_to_fit(edge);
	
	/* ----------------------------------------- Create Output Vertices ------------------------------------------ */
	
	/* Create output polygon */
	out_pgon = Polygon<3, __ndim>(nfaces, pgon.nverts() + edges.nitems());
	
	/* Initialize output vertex iterator */
	auto out_vertex = out_pgon.vbegin();
		
	/* Copy vertices from input to output */
	for_each_item(vertex __in__ pgon.vertices())
	{
		/* Copy vertex */
		*(out_vertex++) = (*vertex);

	} end_for;
	
	/* Calculate new vertices */
	for_each_item(face __in__ edges)
	{
		/* Calculate weighted average of edge vertices */
		*(out_vertex++) = weighted_average(pgon((*face)[0]), pgon((*face)[1]), 
			vcoefs[(*face)[1]] / (vcoefs[(*face)[1]] - vcoefs[(*face)[0]]));

	} end_for;

	/* ------------------------------------------- Create Output Faces ------------------------------------------- */
	
	/* Create face property lists */
	over_plane = Collection<bool>(nfaces);
	original_faces = Collection<Index>(nfaces);
	
	/* Initialize face iterators */
	auto out_face = out_pgon.fbegin();
	auto over = over_plane.begin();
	auto original = original_faces.begin();
	
	/* Copy faces from input to output, splitting faces as necessary */
	for_each_item(face __in__ pgon.faces(), nsplit __in__ nsplits, split __in__ splits)
	{
		/* Determine face index */
		Index findex = (Index)(face - pgon.faces().begin());

		/* Define indexing variables */
		Index i, j, k;

		/* If face doesn't need to be split, copy it to output */
		if (!(*nsplit))
		{
			/* Copy face */
			*(out_face++) = (*face);

			/* Save over / under (defaults to under if all zero) */
			(*over) = false;
			for (Index s = 0; s < 3; ++s)
				if (vcoefs[(*face)[s]] > 0)
					(*over) = true;
			++over;
			
			/* Save original face index */
			*(original++) = findex;
		}
		
		/* Split face in two */
		else if ((*nsplit) == 1)
		{
			/* Locate edge that needs to be split */
			for (i = 0; !(*split)[i]; i++); j = (i + 1) % 3; k = (j + 1) % 3;

			/* Save first face */
			(*out_face)[0] = (*face)[i];
			(*out_face)[1] = (*split)[i];
			(*(out_face++))[2] = (*face)[k];

			/* Save second face */
			(*out_face)[0] = (*split)[i];
			(*out_face)[1] = (*face)[j];
			(*(out_face++))[2] = (*face)[k];

			/* Save over / under */
			*(over++) = vcoefs[(*face)[i]] > 0;
			*(over++) = vcoefs[(*face)[j]] > 0;

			/* Save original face index */
			*(original++) = findex;
			*(original++) = findex;
		}

		/* Split face in three */
		else
		{
			/* Locate edge that remains unsplit */
			for (i = 0; (*split)[i]; i++); j = (i + 1) % 3; k = (j + 1) % 3;
			
			/* Save first face */
			(*out_face)[0] = (*split)[j];
			(*out_face)[1] = (*face)[k];
			(*(out_face++))[2] = (*split)[k];

			/* Save second face */
			(*out_face)[0] = (*face)[j];
			(*out_face)[1] = (*split)[j];
			(*(out_face++))[2] = (*face)[i];

			/* Save third face */
			(*out_face)[0] = (*split)[j];
			(*out_face)[1] = (*split)[k];
			(*(out_face++))[2] = (*face)[i];

			/* Save over / under */
			*(over++) = vcoefs[(*face)[k]] > 0;
			*(over++) = vcoefs[(*face)[i]] > 0;
			*(over++) = vcoefs[(*face)[i]] > 0;

			/* Save original face index */
			*(original++) = findex;
			*(original++) = findex;
			*(original++) = findex;
		}

	} end_for;
	
	/* Return output arguments */
	return std :: tuple<Polygon<3, __ndim>, Collection<bool>, Collection<Index>>(std :: move(out_pgon), 
		std :: move(over_plane), std :: move(original_faces));

	/* ---------------------------------------------- End of Method ---------------------------------------------- */
}
