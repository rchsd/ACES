import numpy as np
import trimesh

def pgonctour(mesh: trimesh.Trimesh, normal: np.ndarray, coefs: np.ndarray):
    """
    Intersects a 3D mesh with multiple parallel planes.

    Parameters:
        mesh: trimesh.Trimesh
            The input mesh (transformed beforehand).
        normal: np.ndarray
            The slicing plane normal (e.g. [0, 0, 1]).
        coefs: np.ndarray
            Offsets along the normal defining each plane.

    Returns:
        contours: List[np.ndarray]
            A list of Nx3 points representing intersection segments.
        face_indices: List[int]
            Indices of mesh faces that contributed to each intersection point.
        plane_indices: List[int]
            Indices of slicing planes corresponding to each intersection point.
    """
    contours = []
    face_indices = []
    plane_indices = []

    vertices = mesh.vertices
    faces = mesh.faces
    vertex_distances = vertices @ normal  # dot product

    for plane_idx, offset in enumerate(coefs):
        signed = vertex_distances - offset

        for face_idx, face in enumerate(faces):
            d = signed[face]
            if np.any(d < 0) and np.any(d > 0):  # intersects the plane
                # loop over edges of the triangle
                for i, j in [(0, 1), (1, 2), (2, 0)]:
                    da, db = d[i], d[j]
                    if da * db < 0:
                        va, vb = vertices[face[i]], vertices[face[j]]
                        w = db / (db - da)
                        p = va * (1 - w) + vb * w
                        contours.append(p)
                        face_indices.append(face_idx)
                        plane_indices.append(plane_idx)

    return np.array(contours), np.array(face_indices), np.array(plane_indices)
